\documentclass[tikz, border=8pt]{standalone}
\usepackage{amsmath}
\usetikzlibrary{positioning, arrows.meta, calc}

\newcommand{\note}[1]{{\small\itshape #1}}
\newcommand{\boxtitle}[1]{{\bfseries #1}}
\newcommand{\eq}[1]{{\small $#1$}}
\newcommand{\boxgap}{0.8cm}
\newcommand{\derivdrop}{0.6cm}
\newcommand{\topclearance}{0.65cm}
\newcommand{\bottomclearance}{2.45cm}
\newcommand{\eqtight}[1]{{\scriptsize $#1$}}

\begin{document}
\begin{tikzpicture}[
    box/.style={
        rectangle,
        draw,
        rounded corners,
        align=center,
        text width=5.1cm,
        minimum height=4.2cm,
        inner sep=6pt,
        anchor=center
    },
    deriv/.style={
        align=center,
        font=\footnotesize,
        text width=4.8cm,
        inner sep=2pt
    },
    >=Stealth
]

    % --- 1) Helmholtz ---
    \node[box] (helm) {%
        \boxtitle{Helmholtz Wave Equation}\\[6pt]
        \eq{\bigl[\nabla^2 + k_0^2\,n^2(\mathbf{r})\bigr]\;\psi(\mathbf{r})=0}\\[12pt]
        \note{Reference full-wave formulation}
    };

    % --- 2) Second-order wave ODE ---
    \node[box, right=\boxgap of helm] (ode) {%
        \boxtitle{Second-Order Wave ODE}\\[6pt]
        \eq{\partial_z^2\psi+\bigl(\nabla_\perp^2+k_0^2 n^2(\mathbf{r})\bigr)\psi=0}\\[12pt]
        \note{Adaptive second-order non-paraxial solve}
    };

    % --- 3) Forward square-root propagator ---
    \node[box, right=\boxgap of ode] (lanczos) {%
        \boxtitle{Forward Square-Root Propagator}\\[6pt]
        \eqtight{\psi(z\!+\!\Delta z)\approx \exp\!\left(i\Delta z\sqrt{\nabla_\perp^2+k_0^2 n^2}\right)\psi(z)}\\[10pt]
        \note{Applied numerically with a Lanczos/Krylov matrix-function approximation}
    };

    % --- 4) WPM (index-dependent spectral step) ---
    \node[box, right=\boxgap of lanczos] (wpm) {%
        \boxtitle{WPM (Wide-angle, local $n$)}\\[6pt]
        \eq{\psi(z\!+\!\Delta z)=\mathcal F^{-1}\!\left[ e^{\,i k_z^{(m)}\Delta z}\mathcal F\{\psi(z)\} \right]}\\[4pt]
        \eq{k_z^{(m)}=\sqrt{(k_0 n_m)^2-k_x^2-k_y^2}}\\[6pt]
        \note{Wide-angle split-step with local refractive index}
    };

    % --- 5) Angular Spectrum (single homogeneous propagator) ---
    \node[box, right=\boxgap of wpm] (as) {%
        \boxtitle{Angular Spectrum (Non-Paraxial)}\\[6pt]
        \eq{\psi(z\!+\!\Delta z)=\mathcal F^{-1}\!\left[ e^{\,i k_z\Delta z}\mathcal F\{\psi(z)\} \right]}\\[4pt]
        \eq{k_z=\sqrt{k^2-k_x^2-k_y^2}}\\[6pt]
        \note{Wide-angle split-step in free space}
    };

    % --- 6) Fresnel (paraxial) ---
    \node[box, right=\boxgap of as] (fres) {%
        \boxtitle{Fresnel (Paraxial)}\\[6pt]
        \eq{\psi(z\!+\!\Delta z)=\mathcal F^{-1}\!\left[ e^{-i\pi\lambda\Delta z k_\perp^2}\mathcal F\{\psi(z)\} \right]}\\[4pt]
        \eq{k_\perp^2=k_x^2+k_y^2}\\[6pt]
        \note{Paraxial split-step in free space}
    };

    % --- Local derivation annotations between methods ---
    \draw[semithick, ->] ($(helm.south east)+(0,-\derivdrop)$) -- ($(ode.south west)+(0,-\derivdrop)$)
        node[midway, below=2pt, deriv]{choose $z$ as propagation axis\\$\nabla^2=\partial_z^2+\nabla_\perp^2$};

    \draw[semithick, ->] ($(ode.south east)+(0,-\derivdrop)$) -- ($(lanczos.south west)+(0,-\derivdrop)$)
        node[midway, below=2pt, deriv]{factor into forward/backward branches\\$\partial_z\psi=i\sqrt{\nabla_\perp^2+k_0^2 n^2}\,\psi$};

    \draw[semithick, ->] ($(lanczos.south east)+(0,-\derivdrop)$) -- ($(wpm.south west)+(0,-\derivdrop)$)
        node[midway, below=2pt, deriv]{evaluate the square-root operator slicewise in Fourier space\\$\sqrt{\nabla_\perp^2+k_0^2 n^2}\to\sqrt{(k_0 n_m)^2-k_x^2-k_y^2}$};

    \draw[semithick, ->] ($(wpm.south east)+(0,-\derivdrop)$) -- ($(as.south west)+(0,-\derivdrop)$)
        node[midway, below=2pt, deriv]{homogeneous-slice limit\\$n(x,y)=\mathrm{const.}$ during the step};

    \draw[semithick, ->] ($(as.south east)+(0,-\derivdrop)$) -- ($(fres.south west)+(0,-\derivdrop)$)
        node[midway, below=2pt, deriv]{small-angle Taylor expansion\\$k_z=\sqrt{k^2-k_\perp^2}\approx k-k_\perp^2/(2k)$};

    % --- Tradeoff arrows (top: faster, bottom: more accurate) ---
    % Calculate exact start and end x-coordinates based on bounding boxes
    \coordinate (start_top) at (helm.north west);
    \coordinate (end_top)   at (fres.north east);
    \coordinate (start_bot) at (helm.south west);
    \coordinate (end_bot)   at (fres.south east);

    \draw[very thick, ->] ($(start_top)+(0,\topclearance)$) -- ($(end_top)+(0,\topclearance)$) 
        node[midway, above]{\Large Faster / simpler $\rightarrow$};
    
    \draw[very thick, <-] ($(start_bot)-(0,\bottomclearance)$) -- ($(end_bot)-(0,\bottomclearance)$) 
        node[midway, below]{\Large $\leftarrow$ More accurate / more general};

\end{tikzpicture}
\end{document}
